using UnityEngine;
using UnityEngine.UI;
using TMPro;
using System.Collections.Generic;
using System.Collections;

public class LabController : MonoBehaviour
{
    [Header("--- 3D Model References ---")]
    public Transform mainModelPivot;
    public GameObject interactiveModeRoot;

    [Header("--- Interactive FX ---")]
    public ParticleSystem successParticles;
    public AudioSource labAudioSource;
    public float typeWriterSpeed = 0.04f;

    [Header("--- Data Definition ---")]
    public List<AnatomyPart> anatomyParts;
    public List<LabStep> interactiveSteps;

    [Header("--- UI References ---")]
    public GameObject leftPanelRoot;
    public GameObject rightPanelRoot;
    public TMP_Dropdown partSelectorDropdown;
    public Slider sliderRotX, sliderRotY, sliderRotZ, sliderZoom;
    public Button btnEnterInteractive, btnExitToMenu;

    [Header("--- Interactive HUD ---")]
    public GameObject interactiveHUDRoot;
    public CanvasGroup interactiveCanvasGroup; 
    public TextMeshProUGUI instructionText;
    public Button btnNextStep;
    public Button btnExitInteractive;

    // Internal State
    private Vector3 initialScale;
    private Quaternion initialRotation;
    private int currentStepIndex = -1;
    private float sessionStartTime;
    private Coroutine currentRoutine;

    [System.Serializable]
    public class AnatomyPart
    {
        public string name;
        public GameObject partObject;
    }

    [System.Serializable]
    public class LabStep
    {
        public string stepName;
        [TextArea] public string instruction;
        public GameObject targetObject;   // The specific model part
        public AudioClip voiceClip;
    }

    [System.Serializable]
    public class LabAnalytics
    {
        public string lastDatePlayed;
        public int highestStepReached;
        public float totalSecondsSpent;
    }

    void Start()
    {
        if (mainModelPivot != null)
        {
            initialScale = mainModelPivot.localScale;
            initialRotation = mainModelPivot.localRotation;
        }

        InitializeDropdown();
        InitializeSliders();
        InitializeButtons();
        SetInteractiveMode(false);
    }

    // ---------------------------------------------------------
    // Setup (Same as before)
    // ---------------------------------------------------------
    void InitializeDropdown()
    {
        if (partSelectorDropdown == null) return;
        partSelectorDropdown.ClearOptions();
        List<string> options = new List<string>();
        foreach (var part in anatomyParts) options.Add(part.name);
        partSelectorDropdown.AddOptions(options);
        partSelectorDropdown.onValueChanged.RemoveAllListeners(); 
        partSelectorDropdown.onValueChanged.AddListener(OnPartSelected);
        if (anatomyParts.Count > 0) { partSelectorDropdown.value = 0; OnPartSelected(0); }
    }

    void InitializeSliders()
    {
        SetupSlider(sliderRotX, 0, 360, (val) => UpdateModelTransform());
        SetupSlider(sliderRotY, 0, 360, (val) => UpdateModelTransform());
        SetupSlider(sliderRotZ, 0, 360, (val) => UpdateModelTransform());
        SetupSlider(sliderZoom, 0.5f, 2.5f, (val) => UpdateModelTransform());
    }

    void SetupSlider(Slider s, float min, float max, UnityEngine.Events.UnityAction<float> action)
    {
        if (s == null) return;
        s.minValue = min; s.maxValue = max;
        s.onValueChanged.RemoveAllListeners(); s.onValueChanged.AddListener(action);
    }

    void InitializeButtons()
    {
        if (btnEnterInteractive != null) btnEnterInteractive.onClick.AddListener(() => SetInteractiveMode(true));
        if (btnExitInteractive != null) btnExitInteractive.onClick.AddListener(() => SetInteractiveMode(false));
        if (btnExitToMenu != null) btnExitToMenu.onClick.AddListener(ExitToMainMenu);
        if (btnNextStep != null) btnNextStep.onClick.AddListener(AdvanceStep);
    }

    void UpdateModelTransform()
    {
        if (mainModelPivot == null) return;
        float x = sliderRotX ? sliderRotX.value : 0;
        float y = sliderRotY ? sliderRotY.value : 0;
        float z = sliderRotZ ? sliderRotZ.value : 0;
        mainModelPivot.localRotation = Quaternion.Euler(x, y, z) * initialRotation;
        mainModelPivot.localScale = initialScale * (sliderZoom ? sliderZoom.value : 1.0f);
    }

    void OnPartSelected(int index)
    {
        for (int i = 0; i < anatomyParts.Count; i++)
        {
            if (anatomyParts[i].partObject != null)
                anatomyParts[i].partObject.SetActive(i == index);
        }
    }

    // =========================================================
    // INTERACTIVE MODE LOGIC (The Fix)
    // =========================================================

    public void SetInteractiveMode(bool isInteractive)
    {
        // Toggle UI
        if (leftPanelRoot) leftPanelRoot.SetActive(!isInteractive);
        if (rightPanelRoot) rightPanelRoot.SetActive(!isInteractive);
        if (mainModelPivot) mainModelPivot.gameObject.SetActive(!isInteractive);
        
        if (interactiveModeRoot) interactiveModeRoot.SetActive(isInteractive);
        if (interactiveHUDRoot) interactiveHUDRoot.SetActive(isInteractive);

        if (isInteractive)
        {
            StartCoroutine(StartInteractiveSession());
        }
        else
        {
            SaveAnalytics();
            StopAllCoroutines();
            if (labAudioSource) labAudioSource.Stop();
        }
    }

    IEnumerator StartInteractiveSession()
    {
        // 1. Fade In HUD
        if (interactiveCanvasGroup != null)
        {
            interactiveCanvasGroup.alpha = 0;
            float time = 0;
            while (time < 0.5f) {
                interactiveCanvasGroup.alpha = Mathf.Lerp(0, 1, time / 0.5f);
                time += Time.deltaTime;
                yield return null;
            }
            interactiveCanvasGroup.alpha = 1;
        }

        // 2. Hide EVERYTHING initially
        foreach(var step in interactiveSteps)
        {
            if(step.targetObject != null) step.targetObject.SetActive(false);
        }

        currentStepIndex = -1;
        sessionStartTime = Time.time;
        if(instructionText) instructionText.text = "";
        
        // 3. Start First Step
        AdvanceStep();
    }

    void AdvanceStep()
    {
        // [FIX] Hide the PREVIOUS model before showing the new one
        if (currentStepIndex >= 0 && currentStepIndex < interactiveSteps.Count)
        {
            var prevStep = interactiveSteps[currentStepIndex];
            if (prevStep.targetObject != null)
            {
                prevStep.targetObject.SetActive(false);
            }
        }

        // Increment Step
        currentStepIndex++;

        // Check if finished
        if (currentStepIndex >= interactiveSteps.Count)
        {
            FinishLab();
            return;
        }

        // Stop old audio/text
        if (currentRoutine != null) StopCoroutine(currentRoutine);
        if (labAudioSource) labAudioSource.Stop();

        // Start new step sequence
        currentRoutine = StartCoroutine(PlayStepSequence(interactiveSteps[currentStepIndex]));
    }

    IEnumerator PlayStepSequence(LabStep step)
    {
        // Disable Next button during intro
        if (btnNextStep) btnNextStep.interactable = false;

        // Ensure target is hidden at start of sequence (reveal it later)
        if (step.targetObject != null) step.targetObject.SetActive(false);

        // 1. Play Audio
        if (labAudioSource != null && step.voiceClip != null)
        {
            labAudioSource.clip = step.voiceClip;
            labAudioSource.Play();
        }

        // 2. Typewriter Text
        if (instructionText != null)
        {
            instructionText.text = "";
            string fullText = step.instruction; // Or add "Step X: " if you want
            foreach (char c in fullText)
            {
                instructionText.text += c;
                yield return new WaitForSeconds(typeWriterSpeed);
            }
        }

        // Small pause for effect
        yield return new WaitForSeconds(0.2f);

        // 3. REVEAL MODEL
        if (step.targetObject != null)
        {
            step.targetObject.SetActive(true);

            // 4. Play Particles (Force Restart)
            if (successParticles != null)
            {
                successParticles.transform.position = step.targetObject.transform.position;
                
                // STOP first to clear old particles, then PLAY
                successParticles.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);
                successParticles.Play(); 
            }
        }

        // 5. Re-enable Next Button (User can now proceed)
        if (btnNextStep) btnNextStep.interactable = true;
    }

    void FinishLab()
    {
        if (instructionText) instructionText.text = "Lab Completed! Saving Data...";
        if (btnNextStep) btnNextStep.interactable = false;
        SaveAnalytics();
        Invoke("ExitToMainMenu", 2.0f);
    }

    void SaveAnalytics()
    {
        string json = PlayerPrefs.GetString("LabAnalytics", "");
        LabAnalytics data = new LabAnalytics();
        if (!string.IsNullOrEmpty(json)) data = JsonUtility.FromJson<LabAnalytics>(json);

        data.lastDatePlayed = System.DateTime.Now.ToString();
        data.totalSecondsSpent += (Time.time - sessionStartTime);
        if (currentStepIndex > data.highestStepReached) data.highestStepReached = currentStepIndex;

        PlayerPrefs.SetString("LabAnalytics", JsonUtility.ToJson(data));
        PlayerPrefs.Save();
    }

    void ExitToMainMenu()
    {
        if (SceneChanger.Instance != null) SceneChanger.Instance.ChangeScene("MainMenu"); 
        else UnityEngine.SceneManagement.SceneManager.LoadScene("MainMenu");
    }
}
